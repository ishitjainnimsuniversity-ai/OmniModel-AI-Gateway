2. Standard OpenAI Python SDK 
from openai import OpenAI

# Connect to local OmniModel Gateway
client = OpenAI(
    base_url="http://localhost:8000/v1",
    api_key="omnimodel-local" # any placeholder key
)

# Call smart auto-router or any specific model across 20 providers:
response = client.chat.completions.create(
    model="auto", # or "groq/llama-3.3-70b-versatile", "gemini/gemini-2.0-flash", etc.
    messages=[{"role": "user", "content": "What is the capital of France?"}],
    stream=True
)

for chunk in response:
    content = chunk.choices[0].delta.content or ""
    print(content, end="", flush=True)
3. JavaScript / Node.js SDK 
import OpenAI from "openai";

const openai = new OpenAI({
  baseURL: "http://localhost:8000/v1",
  apiKey: "omnimodel-local",
});

async function main() {
  const completion = await openai.chat.completions.create({
    model: "free_tier", // Routes to verified free endpoints
    messages: [{ role: "user", content: "Hello from OmniModel Gateway!" }],
  });
  console.log(completion.choices[0].message.content);
}
main();
