2. Standard OpenAI Python SDK 
from openai import OpenAI

# Connect to local GENESIS AI 5.0 Gateway
client = OpenAI(
    base_url="http://localhost:8000/v1",
    api_key="genesis-ai-local"
)

# Query flagship virtual models or any specific provider:
response = client.chat.completions.create(
    model="genesis-ai-5.0", # or "genesis-5.0-reasoning", "genesis-5.0-free", etc.
    messages=[{"role": "user", "content": "How does self-attention work?"}],
    stream=True
)

for chunk in response:
    content = chunk.choices[0].delta.content or ""
    print(content, end="", flush=True)
3. JavaScript / Node.js SDK 
import OpenAI from "openai";

const openai = new OpenAI({
  baseURL: "http://localhost:8000/v1",
  apiKey: "genesis-ai-local",
});

async function main() {
  const completion = await openai.chat.completions.create({
    model: "genesis-5.0-free",
    messages: [{ role: "user", content: "Hello GENESIS AI 5.0!" }],
  });
  console.log(completion.choices[0].message.content);
}
main();
